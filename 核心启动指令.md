# 核心启动指令

## NUC 宿主机：普通室外

```bash
cd /home/nuc15/lxw/neupan_ws
bash src/robot_base_driver/scripts/start_outdoor.sh \
  neupan_result_publish_timeout:=0.60 neupan_result_compute_timeout:=0.50 \
  enable_camera_right_path:=false enable_camera_right_control:=false
```

## NUC 宿主机：摄像头靠右

```bash
cd /home/nuc15/lxw/neupan_ws
bash src/robot_base_driver/scripts/start_outdoor.sh \
  neupan_result_publish_timeout:=0.60 neupan_result_compute_timeout:=0.50 \
  enable_camera_right_path:=true enable_camera_right_control:=true \
  camera_right_strategy:=segment_reference \
  camera_right_camera_clearance_m:=2.0 \
  camera_right_clearance_min_m:=1.0 camera_right_clearance_max_m:=3.0 \
  camera_right_base_link_height_m:=0.20 \
  enable_system_log:=true enable_hybridnets_overlay:=false enable_rviz:=false
```

## GPU：摄像头推流

```bash
conda deactivate
cd /home/p/lxw/lxwgdutcar_ai_ws
source /opt/ros/humble/setup.bash
source install/setup.bash
ros2 launch usb_camera hybridnets_stream.launch.py \
  backend:=tensorrt target_ip:=192.168.1.10 \
  video_port:=5000 result_port:=6000 bitrate:=8000000
```

## NUC 宿主机：靠右录包（另开终端）

```bash
docker exec -it Neupan bash -lc '
source /root/miniconda3/etc/profile.d/conda.sh
conda activate lxw
source /opt/ros/noetic/setup.bash
source /home/nuc/lxw/neupan_ws/devel/setup.bash
cd /home/nuc/lxw/neupan_ws
roslaunch vehicle_bag_recorder record.launch \
  config_file:=$(rospack find vehicle_bag_recorder)/config/camera_right_test.yaml
'
```

## NUC 宿主机：核对实际参数

```bash
docker exec -it Neupan bash -lc '
source /opt/ros/noetic/setup.bash
rosparam list | grep -E "transport_limit|common_utc|raw_capture|low_precision|result_publish_timeout|result_compute_timeout|auto_timeout" | while read -r key; do
  printf "%s = " "$key"
  rosparam get "$key"
done
rostopic echo /gps_guard/status
'
```
