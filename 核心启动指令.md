# 核心启动指令

## ROS1 终端：普通室外

```bash
cd /home/nuc/lxw/neupan_ws
conda activate lxw
source /opt/ros/noetic/setup.bash
source /home/nuc/lxw/neupan_ws/devel/setup.bash
roslaunch neupan_ros full_system_run.launch \
  outdoor_mode:=true low_precision_startup:=true \
  gps_transport_limit_sec:=4.0 gps_common_utc_verified:=true gps_raw_capture_sec:=60 \
  neupan_result_publish_timeout:=0.60 neupan_result_compute_timeout:=0.50 \
  enable_camera_right_path:=false enable_camera_right_control:=false \
  enable_hybridnets_overlay:=false
```

## ROS1 终端：核对实际参数

```bash
source /opt/ros/noetic/setup.bash
rosparam list | grep -E "transport_limit|common_utc|raw_capture|low_precision|result_publish_timeout|result_compute_timeout|auto_timeout" | while read -r key; do
  printf "%s = " "$key"
  rosparam get "$key"
done
rostopic echo /gps_guard/status
```
